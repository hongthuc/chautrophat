        function handleImgError(imgEl, driveId, labelText) {
            const step = parseInt(imgEl.dataset.errStep || '0', 10);
            if (step === 0 && driveId) {
                imgEl.dataset.errStep = '1';
                imgEl.src = 'https://lh3.googleusercontent.com/d/' + driveId + '=w1000';
            } else if (step === 1 && driveId) {

                imgEl.src = 'https://drive.google.com/uc?export=view&id=' + driveId;

                imgEl.onerror = null;
                imgEl.src = 'https://placehold.co/600x450/0B3D6E/FFFFFF?text=' + encodeURIComponent(labelText.replace(/\+/g, ' '));

        /* =========================================================

        let currentBannerIdx = 0;
        let bannerTimer = null;

        function updateBannerUI() {
            const slides = document.querySelectorAll('#hero-slider .banner-slide');
            slides.forEach((slide, i) => {
                slide.classList.toggle('is-active', i === currentBannerIdx);
                const dot = document.getElementById('banner-dot-' + i);
                if (dot) {
                    dot.className = i === currentBannerIdx
                        ? 'w-6 h-2 rounded-full bg-brand-accent transition-all'
                        : 'w-2.5 h-2 rounded-full bg-white/50 hover:bg-white transition-all';
                }
            });
        }

        function nextBanner() {
            currentBannerIdx = (currentBannerIdx + 1) % 2;
            updateBannerUI();
            resetBannerTimer();
        }

        function prevBanner() {
            currentBannerIdx = (currentBannerIdx + 1) % 2;
            updateBannerUI();
            resetBannerTimer();
        }

        function goToBanner(idx) {
            currentBannerIdx = idx;
            updateBannerUI();
            resetBannerTimer();
        }

        function resetBannerTimer() {
            if (bannerTimer) clearInterval(bannerTimer);
            bannerTimer = setInterval(() => {
                currentBannerIdx = (currentBannerIdx + 1) % 2;
                updateBannerUI();
            }, 5000);
        }

           XỬ LÝ FORM YÊU CẦU BÁO GIÁ (Không dùng alert)
        ========================================================== */
        function prefillQuoteCategory(catText) {
            const sel = document.getElementById('q-category');
            const note = document.getElementById('q-note');
            if (sel) {
                let matched = false;
                for (let i = 0; i < sel.options.length; i++) {
                    if (sel.options[i].value.toLowerCase() === catText.toLowerCase()) {
                        sel.selectedIndex = i;
                        matched = true;
                        break;
                    }
                }
                if (!matched && note) {
                    note.value = 'Tôi cần tư vấn và báo giá mẫu: ' + catText;
                }
            }
        }
